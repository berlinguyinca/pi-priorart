# priorart Implementation Plan — Phase 0 + Phase 1

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `priorart`, a Go MCP server + portable protocol + synced decision ledger that forces every harness (codex/claude/pi/opencode) to search for prior art (own repos + public OSS) before writing non-trivial new code.

**Architecture:** A zero-dependency Go static binary exposes an MCP stdio server. It fans a query out to pluggable search backends (npm, libraries.io, crates.io, GitHub, grep.app, local ripgrep), ranks the merged results, consults/appends a synced JSONL decision ledger, and returns a structured result. A canonical "reuse-first" protocol is synced by ai-sync into each harness's instruction file.

**Tech Stack:** Go 1.25, stdlib only (`net/http`, `encoding/json`, `os/exec` for ripgrep, `io/fs`). No external modules. Bash for `install.sh`.

**Spec:** `docs/superpowers/specs/2026-11-01-priorart-design.md`

## Global Constraints

- Module path: `github.com/berlinguyinca/priorart`. All internal imports use it.
- **Zero external dependencies.** `go.mod` has no `require` lines. Use stdlib only.
- Build: `CGO_ENABLED=0 go build -ldflags "-s -w -extldflags '-static'"` → single static binary named `priorart`.
- Go 1.25 (toolchain already present at `/usr/local/go/bin/go`).
- **Secrets from env only** (`GITHUB_TOKEN`, `CONTEXT7_API_KEY`). Never read or persisted in `priorart.json`. Config struct uses `json:"-"` for them.
- **Every MCP tool call returns a structured result and never hard-fails the agent's turn.** A backend error → recorded in `skipped`, not an error return that aborts.
- **Search results are untrusted data.** Snippets are truncated to 300 chars; the server never executes anything derived from a result.
- One file per backend, all implementing `backends.Backend` (see Task 5).
- Conventional commits (`feat:`, `test:`, `chore:`, `docs:`, `fix:`). Commit after each task.
- All tests must pass with `CGO_ENABLED=0 go test ./...`.
- **Spec deviation (approved):** `docs` is NOT reimplemented inside `priorart`. Doc lookup is delegated to the native Context7 MCP already configured in every harness (re-enabled on opencode in Task 1). The protocol references "Context7" generically.
- **Spec deviation (approved):** Go OSS search is served by the GitHub backend with a `language:Go` query qualifier (no separate `gosrc` backend), per spec §19 (Go is the weakest OSS target).

```go
// Package candidate defines the shared data types returned by priorart.
package candidate

// Candidate is a single prior-art hit from any backend.
type Candidate struct {
	Source      string   `json:"source"` // "own" | "oss"
	Ecosystem   string   `json:"ecosystem"`
	Name        string   `json:"name"`
	Version     string   `json:"version,omitempty"`
	RepoURL     string   `json:"repo_url,omitempty"`
	License     string   `json:"license,omitempty"`
	Activity    Activity `json:"activity,omitempty"`
	Snippet     string   `json:"snippet,omitempty"`
	MatchReason string   `json:"match_reason,omitempty"`
	DocsURL     string   `json:"docs_url,omitempty"`
}

type Activity struct {
	LastRelease string `json:"last_release,omitempty"`
	Stars       int    `json:"stars,omitempty"`
}

// PriorArtResult is the return of the prior_art tool.
type PriorArtResult struct {
	Query      string      `json:"query"`
	Decision   *Decision   `json:"prior_decision,omitempty"`
	Candidates []Candidate `json:"candidates"`
	Searched   []string    `json:"searched"`
	Skipped    []string    `json:"skipped,omitempty"`
}

// Decision is a recorded adopt-vs-build choice (ledger line).
type Decision struct {
	Key                  string   `json:"key"`
	Query                string   `json:"query"`
	Scope                string   `json:"scope,omitempty"`
	Action               string   `json:"action"` // "adopt" | "build"
	Choice               string   `json:"choice"`
	Reason               string   `json:"reason"`
	CandidatesConsidered []string `json:"candidates_considered,omitempty"`
	TS                   string   `json:"ts"`
	Machine              string   `json:"machine,omitempty"`
}

// Wheel is a "known wheel" that must not be reimplemented.
type Wheel struct {
	Name        string   `json:"name"`
	Description string   `json:"description"`
	Canonical   []string `json:"canonical"`
}
```

```go
// Package config loads priorart configuration. Secrets come from env only.
package config

import (
	"encoding/json"
	"errors"
	"io/fs"
	"os"
)

type GitHub struct {
	OwnAccount  string `json:"own_account"`
	UseOwnLayer bool   `json:"use_own_layer"`
}

type Context7 struct {
	Enabled bool `json:"enabled"`
}

type Config struct {
	OwnRoots        []string  `json:"own_roots"`
	OwnSSH          []string  `json:"own_ssh"`
	Ecosystems      []string  `json:"ecosystems"`
	GitHub          GitHub    `json:"github"`
	LedgerPath      string    `json:"ledger_path"`
	Context7        Context7  `json:"context7"`
	BlockLicenses   []string  `json:"block_licenses"`
	CacheTTLSeconds int       `json:"cache_ttl_seconds"`

	GitHubToken    string `json:"-"` // env: GITHUB_TOKEN
	Context7APIKey string `json:"-"` // env: CONTEXT7_API_KEY
}

func Default() Config {
	return Config{
		Ecosystems:      []string{"py", "js", "rs", "go"},
		BlockLicenses:   []string{"GPL-3.0", "AGPL-3.0"},
		CacheTTLSeconds: 3600,
		Context7:        Context7{Enabled: true},
	}
}

func Load(path string) (Config, error) {
	cfg := Default()
	if path != "" {
		b, err := os.ReadFile(path)
		if err != nil {
			if errors.Is(err, fs.ErrNotExist) {
				return cfg, nil // defaults + env only
			}
			return cfg, err
		}
		if err := json.Unmarshal(b, &cfg); err != nil {
			return cfg, err
		}
	}
	cfg.GitHubToken = os.Getenv("GITHUB_TOKEN")
	cfg.Context7APIKey = os.Getenv("CONTEXT7_API_KEY")
	return cfg, nil
}
```

```go
// Package ledger stores adopt-vs-build decisions as synced JSONL.
package ledger

import (
	"crypto/sha1"
	"encoding/hex"
	"encoding/json"
	"os"
	"path/filepath"
	"strings"
	"time"

	"github.com/berlinguyinca/priorart/internal/candidate"
)

func Normalize(query, scope string) string {
	s := strings.ToLower(strings.TrimSpace(query))
	s = strings.Join(strings.Fields(s), " ")
	if scope != "" {
		s = scope + "|" + s
	}
	sum := sha1.Sum([]byte(s))
	return hex.EncodeToString(sum[:])
}

func Lookup(path, key string) (*candidate.Decision, bool) {
	f, err := os.Open(path)
	if err != nil {
		return nil, false
	}
	defer f.Close()
	dec := json.NewDecoder(f)
	var best *candidate.Decision
	for {
		var d candidate.Decision
		if err := dec.Decode(&d); err != nil {
			break
		}
		if d.Key == key {
			cp := d
			best = &cp
		}
	}
	if best != nil {
		return best, true
	}
	return nil, false
}

func Append(path string, d candidate.Decision) error {
	if d.Key == "" {
		d.Key = Normalize(d.Query, d.Scope)
	}
	if d.TS == "" {
		d.TS = time.Now().UTC().Format(time.RFC3339)
	}
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		return err
	}
	f, err := os.OpenFile(path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
	if err != nil {
		return err
	}
	defer f.Close()
	b, err := json.Marshal(d)
	if err != nil {
		return err
	}
	_, err = f.Write(append(b, '\n'))
	return err
}
```

```go
// Package npm searches the npm registry for JS/TS prior art.
package npm

import (
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"net/url"
	"time"

	"github.com/berlinguyinca/priorart/internal/candidate"
)

type Client struct {
	Base string
	HTTP *http.Client
}

func New(baseURL string, httpc *http.Client) *Client {
	if baseURL == "" {
		baseURL = "https://registry.npmjs.org"
	}
	if httpc == nil {
		httpc = &http.Client{Timeout: 10 * time.Second}
	}
	return &Client{Base: baseURL, HTTP: httpc}
}

func (c *Client) Name() string { return "npm" }

type npmSearchResponse struct {
	Objects []struct {
		Package struct {
			Name        string `json:"name"`
			Version     string `json:"version"`
			Description string `json:"description"`
			Links       struct {
				Repository string `json:"repository"`
				Homepage   string `json:"homepage"`
			} `json:"links"`
		} `json:"package"`
	} `json:"objects"`
}

func (c *Client) Search(ctx context.Context, query, _ string, limit int) ([]candidate.Candidate, error) {
	if limit <= 0 {
		limit = 5
	}
	u := fmt.Sprintf("%s/-/v1/search?text=%s&size=%d", c.Base, url.QueryEscape(query), limit)
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, u, nil)
	if err != nil {
		return nil, err
	}
	resp, err := c.HTTP.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
```

```go
// Package github searches GitHub repositories for prior art.
package github

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"net/http"
	"net/url"
	"time"

	"github.com/berlinguyinca/priorart/internal/candidate"
)

type Client struct {
	Base  string
	Token string
	HTTP  *http.Client
}

func New(baseURL, token string, httpc *http.Client) *Client {
	if baseURL == "" {
		baseURL = "https://api.github.com"
	}
	if httpc == nil {
		httpc = &http.Client{Timeout: 10 * time.Second}
	}
	return &Client{Base: baseURL, Token: token, HTTP: httpc}
}

func (c *Client) Name() string { return "github" }

type repoSearchResponse struct {
	Items []struct {
		FullName        string `json:"full_name"`
		HTMLURL         string `json:"html_url"`
		Description     string `json:"description"`
		StargazersCount int    `json:"stargazers_count"`
		License         *struct {
			SpdxID string `json:"spdx_id"`
		} `json:"license"`
		Language string `json:"language"`
		PushedAt string `json:"pushed_at"`
	} `json:"items"`
}

func (c *Client) do(ctx context.Context, q string, limit int) (*repoSearchResponse, error) {
	if c.Token == "" {
		return nil, errors.New("github: GITHUB_TOKEN not set")
	}
	u := fmt.Sprintf("%s/search/repositories?q=%s&per_page=%d", c.Base, url.QueryEscape(q), limit)
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, u, nil)
	if err != nil {
		return nil, err
	}
	req.Header.Set("Authorization", "Bearer "+c.Token)
	req.Header.Set("Accept", "application/vnd.github+json")
	resp, err := c.HTTP.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("github: status %d", resp.StatusCode)
	}
	var r repoSearchResponse
	if err := json.NewDecoder(resp.Body).Decode(&r); err != nil {
		return nil, err
	}
	return &r, nil
}

func (c *Client) toCandidates(r *repoSearchResponse, eco string) []candidate.Candidate {
	out := make([]candidate.Candidate, 0, len(r.Items))
	for _, it := range r.Items {
		lic := ""
		if it.License != nil {
			lic = it.License.SpdxID
		}
		out = append(out, candidate.Candidate{
			Source: "oss", Ecosystem: eco,
			Name: it.FullName, RepoURL: it.HTMLURL, License: lic,
			Activity:    candidate.Activity{LastRelease: it.PushedAt, Stars: it.StargazersCount},
			Snippet:     it.Description, MatchReason: "github repo search match", DocsURL: it.HTMLURL,
		})
	}
	return out
}

// Search runs a repository search; when ecosystem is "go" it scopes to Go.
func (c *Client) Search(ctx context.Context, query, ecosystem string, limit int) ([]candidate.Candidate, error) {
	if limit <= 0 {
		limit = 5
	}
	q, eco := query, "oss"
	if ecosystem == "go" {
		q = query + " language:Go"
		eco = "go"
	}
	r, err := c.do(ctx, q, limit)
	if err != nil {
		return nil, err
	}
	return c.toCandidates(r, eco), nil
}

// SearchGo is an explicit Go-only repository search.
func (c *Client) SearchGo(ctx context.Context, query string, limit int) ([]candidate.Candidate, error) {
	if limit <= 0 {
		limit = 5
	}
	r, err := c.do(ctx, query+" language:Go", limit)
	if err != nil {
		return nil, err
	}
	return c.toCandidates(r, "go"), nil
}
```

```go
type Client struct {
	Roots  []string
	RGPath string
}

func New(roots []string, rgPath string) *Client {
	if rgPath == "" {
		rgPath = "rg"
	}
	return &Client{Roots: roots, RGPath: rgPath}
}

func (c *Client) Name() string { return "own-local" }

var tokenRe = regexp.MustCompile(`[a-zA-Z][a-zA-Z0-9_]{3,}`)

// Tokens extracts search tokens (alphanumeric, len>=4) from a query, lowercased, deduped.
func Tokens(query string) []string {
	seen := map[string]bool{}
	var out []string
	for _, m := range tokenRe.FindAllString(strings.ToLower(query), -1) {
		if !seen[m] {
			seen[m] = true
			out = append(out, m)
		}
	}
	return out
}

const snippetMax = 300

func clip(s string) string {
	s = strings.TrimSpace(s)
	if len(s) > snippetMax {
		return s[:snippetMax]
	}
	return s
}

type hit struct {
	path    string
	snippet string
	token   string
}

func (c *Client) Search(ctx context.Context, query, _ string, limit int) ([]candidate.Candidate, error) {
	if limit <= 0 {
		limit = 10
	}
	toks := Tokens(query)
	if len(toks) == 0 {
		return nil, nil
	}
	rgOK := exec.LookPath(c.RGPath) == nil
	var hits []hit
	seen := map[string]bool{}
	add := func(h hit) {
		if !seen[h.path] {
			seen[h.path] = true
			hits = append(hits, h)
		}
	}
	for _, root := range c.Roots {
		if _, err := os.Stat(root); err != nil {
			continue
		}
		if rgOK {
			for _, h := range c.ripgrep(ctx, root, toks, limit) {
				add(h)
			}
			continue
		}
		for _, h := range c.walk(root, toks, limit) {
			add(h)
		}
	}
	out := make([]candidate.Candidate, 0, len(hits))
	for _, h := range hits {
		if len(out) >= limit {
			break
		}
		out = append(out, candidate.Candidate{
			Source: "own", Name: "own:" + h.path, Snippet: clip(h.snippet),
			MatchReason: "own repo match (token " + h.token + ")",
		})
	}
	return out, nil
}

func (c *Client) ripgrep(ctx context.Context, root string, toks []string, limit int) []hit {
	var out []hit
	for _, tok := range toks {
		args := []string{"-n", "--no-heading", "-F", "-m", "1", tok, root}
		cmd := exec.CommandContext(ctx, c.RGPath, args...)
		b, err := cmd.Output()
		if err != nil {
			continue // rg exits 1 on no match; treat as empty
		}
		sc := bufio.NewScanner(strings.NewReader(string(b)))
		sc.Buffer(make([]byte, 0, 64*1024), 1024*1024)
		for sc.Scan() {
			line := sc.Text()
			parts := strings.SplitN(line, ":", 3)
			if len(parts) < 3 {
				continue
			}
			out = append(out, hit{path: parts[0], snippet: parts[2], token: tok})
			if len(out) >= limit*len(toks) {
				return out
			}
		}
	}
	return out
}

var skipDir = map[string]bool{".git": true, "node_modules": true, "vendor": true, "dist": true, "build": true, ".venv": true, "venv": true}

func (c *Client) walk(root string, toks []string, limit int) []hit {
	var out []hit
	_ = filepath.WalkDir(root, func(p string, d fs.DirEntry, err error) error {
		if err != nil {
			return nil
		}
		if d.IsDir() {
			if skipDir[d.Name()] {
				return filepath.SkipDir
			}
			return nil
		}
		f, err := os.Open(p)
		if err != nil {
			return nil
		}
		defer f.Close()
		sc := bufio.NewScanner(f)
		sc.Buffer(make([]byte, 0, 64*1024), 1024*1024)
		for sc.Scan() {
			line := sc.Text()
			low := strings.ToLower(line)
			for _, tok := range toks {
				if strings.Contains(low, tok) {
					out = append(out, hit{path: p, snippet: line, token: tok})
					if len(out) >= limit*len(toks) {
						return nil
					}
					break
				}
			}
		}
		return nil
	})
	return out
}
```

```go
// Package rank merges, dedups, and scores prior-art candidates.
package rank

import (
	"sort"
	"strings"

	"github.com/berlinguyinca/priorart/internal/candidate"
	"github.com/berlinguyinca/priorart/internal/config"
	"github.com/berlinguyinca/priorart/internal/wheels"
)

// Rank dedups by Name, drops blocked licenses, scores, and sorts descending.
func Rank(cands []candidate.Candidate, query string, cfg config.Config) []candidate.Candidate {
	block := map[string]bool{}
	for _, l := range cfg.BlockLicenses {
		block[strings.ToLower(l)] = true
	}
	best := map[string]candidate.Candidate{}
	for _, c := range cands {
		if c.Name == "" {
			continue
		}
		if c.License != "" && block[strings.ToLower(c.License)] {
			continue
		}
		if prev, ok := best[c.Name]; !ok || score(c, query) > score(prev, query) {
			best[c.Name] = c
		}
	}
	out := make([]candidate.Candidate, 0, len(best))
	for _, c := range best {
		out = append(out, c)
	}
	sort.SliceStable(out, func(i, j int) bool {
		si, sj := score(out[i], query), score(out[j], query)
		if si != sj {
			return si > sj
		}
		return out[i].Name < out[j].Name
	})
	return out
}

func score(c candidate.Candidate, query string) float64 {
	s := 0.0
	if c.Source == "own" {
		s += 3.0
	}
	if q := strings.ToLower(query); q != "" && strings.Contains(strings.ToLower(c.Name), q) {
		s += 1.0
	}
	for _, w := range wheels.All() {
		for _, canon := range w.Canonical {
			if strings.EqualFold(canon, c.Name) {
				s += 5.0
			}
		}
	}
	if c.Activity.Stars > 0 {
		s += 1.0
	}
	return s
}
```

```go
// Package index builds a deterministic per-repo capability index.
package index

import (
	"bufio"
	"encoding/json"
	"os"
	"path/filepath"
	"sort"
	"strings"
)

type RepoEntry struct {
	Repo    string   `json:"repo"`
	Title   string   `json:"title,omitempty"`
	Summary string   `json:"summary,omitempty"`
	Files   []string `json:"files"`
}

const maxFiles = 200

// Build indexes every immediate subdirectory of root that looks like a repo.
func Build(root, outDir string) ([]RepoEntry, error) {
	if err := os.MkdirAll(outDir, 0o755); err != nil {
		return nil, err
	}
	dirs, err := os.ReadDir(root)
	if err != nil {
		return nil, err
	}
	var out []RepoEntry
	for _, d := range dirs {
		if !d.IsDir() || strings.HasPrefix(d.Name(), ".") {
			continue
		}
		e := buildOne(d.Name(), filepath.Join(root, d.Name()))
		b, _ := json.Marshal(e)
		_ = os.WriteFile(filepath.Join(outDir, d.Name()+".json"), b, 0o644)
		out = append(out, e)
	}
	sort.Slice(out, func(i, j int) bool { return out[i].Repo < out[j].Repo })
	return out, nil
}

func buildOne(repo, repoPath string) RepoEntry {
	e := RepoEntry{Repo: repo}
	if readme := readReadme(repoPath); readme != "" {
		e.Title, e.Summary = parseReadme(readme)
	}
	var files []string
	_ = filepath.WalkDir(repoPath, func(p string, d os.DirEntry, err error) error {
		if err != nil {
			return nil
		}
		if d.IsDir() {
			switch d.Name() {
			case ".git", "node_modules", "vendor", "dist", "build", ".venv", "venv":
				return filepath.SkipDir
			}
			return nil
		}
		if len(files) < maxFiles && isSource(d.Name()) {
			rel, _ := filepath.Rel(repoPath, p)
			files = append(files, rel)
		}
		return nil
	})
	sort.Strings(files)
	e.Files = files
	return e
}

func readReadme(repoPath string) string {
	for _, n := range []string{"README.md", "README.rst", "README"} {
		if b, err := os.ReadFile(filepath.Join(repoPath, n)); err == nil {
			return string(b)
		}
	}
	return ""
}

func parseReadme(s string) (title, summary string) {
	sc := bufio.NewScanner(strings.NewReader(s))
	sc.Buffer(make([]byte, 0, 64*1024), 1024*1024)
	for sc.Scan() {
		line := strings.TrimSpace(sc.Text())
		if title == "" && strings.HasPrefix(line, "# ") {
			title = strings.TrimPrefix(line, "# ")
			continue
		}
		if title != "" && summary == "" && line != "" && !strings.HasPrefix(line, "#") {
			summary = line
			break
		}
	}
	return
}

func isSource(name string) bool {
	switch strings.ToLower(name) {
	case ".go", ".py", ".ts", ".js", ".mjs", ".rs", ".java", ".scala", ".rb", ".c", ".cpp", ".cs", ".sh":
		return true
	}
	return false
}

// Load reads a previously built entry.
func Load(outDir, repo string) (RepoEntry, bool) {
	var e RepoEntry
	if b, err := os.ReadFile(filepath.Join(outDir, repo+".json")); err == nil {
		_ = json.Unmarshal(b, &e)
		return e, true
	}
	return e, false
}
```

```go
// Package core orchestrates ledger lookup, backend fan-out, ranking, and caching.
package core

import (
	"context"
	"sync"
	"time"

	"github.com/berlinguyinca/priorart/internal/backends"
	"github.com/berlinguyinca/priorart/internal/candidate"
	"github.com/berlinguyinca/priorart/internal/config"
	"github.com/berlinguyinca/priorart/internal/ledger"
	"github.com/berlinguyinca/priorart/internal/rank"
)

type cacheEntry struct {
	cats []candidate.Candidate
	at   time.Time
}

type Core struct {
	cfg      config.Config
	backends []backends.Backend
	ledgerP  string
	mu       sync.Mutex
	cache    map[string]cacheEntry
	CacheTTL int
}

func New(cfg config.Config, bs []backends.Backend) *Core {
	return &Core{cfg: cfg, backends: bs, cache: map[string]cacheEntry{}, CacheTTL: cfg.CacheTTLSeconds}
}

func (c *Core) SetLedgerPath(p string) { c.ledgerP = p }

func (c *Core) PriorArt(ctx context.Context, query, ecosystem string, limit int) (candidate.PriorArtResult, error) {
	if limit <= 0 {
		limit = 5
	}
	res := candidate.PriorArtResult{Query: query, Searched: []string{}}
	key := ledger.Normalize(query, ecosystem)
	if d, ok := ledger.Lookup(c.ledgerP, key); ok {
		res.Decision = d
	}

	ckey := ecosystem + "|" + key
	if c.CacheTTL > 0 {
		c.mu.Lock()
		if ce, ok := c.cache[ckey]; ok && time.Since(ce.at) < time.Duration(c.CacheTTL)*time.Second {
			res.Candidates = rank.Rank(ce.cats, query, c.cfg)
			c.mu.Unlock()
			return res, nil
		}
		c.mu.Unlock()
	}

	var merged []candidate.Candidate
	for _, b := range c.backends {
		cats, err := b.Search(ctx, query, ecosystem, limit)
		if err != nil {
			res.Skipped = append(res.Skipped, b.Name()+": "+err.Error())
			continue
		}
		merged = append(merged, cats...)
		res.Searched = append(res.Searched, b.Name())
	}
	ranked := rank.Rank(merged, query, c.cfg)
	if c.CacheTTL > 0 {
		c.mu.Lock()
		c.cache[ckey] = cacheEntry{cats: ranked, at: time.Now()}
		c.mu.Unlock()
	}
	res.Candidates = ranked
	return res, nil
}

func (c *Core) SearchOwn(ctx context.Context, query string, limit int) ([]candidate.Candidate, error) {
	for _, b := range c.backends {
		if b.Name() == "own-local" {
			return b.Search(ctx, query, "", limit)
		}
	}
	return nil, nil
}

func (c *Core) RecordDecision(d candidate.Decision) error {
	return ledger.Append(c.ledgerP, d)
}
```

```go
// Package mcp implements a minimal MCP stdio server (JSON-RPC 2.0).
package mcp

import (
	"bufio"
	"context"
	"encoding/json"
	"fmt"
	"io"

	"github.com/berlinguyinca/priorart/internal/candidate"
	"github.com/berlinguyinca/priorart/internal/core"
	"github.com/berlinguyinca/priorart/internal/wheels"
)

type RPCError struct {
	Code    int    `json:"code"`
	Message string `json:"message"`
}

func (e *RPCError) Error() string { return fmt.Sprintf("rpc %d: %s", e.Code, e.Message) }

type Server struct{ core *core.Core }

func New(c *core.Core) *Server { return &Server{core: c} }

type toolDef struct {
	Name        string         `json:"name"`
	Description string         `json:"description"`
	InputSchema map[string]any `json:"inputSchema"`
}

func schema(props map[string]any, req ...string) map[string]any {
	return map[string]any{"type": "object", "properties": props, "required": req}
}

func (s *Server) tools() []toolDef {
	return []toolDef{
		{Name: "prior_art", Description: "Search own repos + public OSS for prior art before writing code.",
			InputSchema: schema(map[string]any{
				"query":     map[string]any{"type": "string"},
				"ecosystem": map[string]any{"type": "string"},
				"limit":     map[string]any{"type": "integer"},
			}, "query")},
		{Name: "search_own", Description: "Search only your own repos.",
			InputSchema: schema(map[string]any{"query": map[string]any{"type": "string"}, "limit": map[string]any{"type": "integer"}}, "query")},
		{Name: "search_registry", Description: "Search one package registry.",
			InputSchema: schema(map[string]any{
				"ecosystem": map[string]any{"type": "string", "enum": []string{"py", "js", "rs", "go"}},
				"query":     map[string]any{"type": "string"},
				"limit":     map[string]any{"type": "integer"},
			}, "ecosystem", "query")},
		{Name: "record_decision", Description: "Record an adopt-vs-build decision in the ledger.",
			InputSchema: schema(map[string]any{
				"query":  map[string]any{"type": "string"},
				"action": map[string]any{"type": "string", "enum": []string{"adopt", "build"}},
				"choice": map[string]any{"type": "string"},
				"reason": map[string]any{"type": "string"},
			}, "query", "action", "choice", "reason")},
		{Name: "wheels", Description: "List known wheels that must not be reimplemented.",
			InputSchema: schema(map[string]any{})},
	}
}

func (s *Server) Dispatch(method string, params map[string]any) (any, *RPCError) {
	switch method {
	case "initialize":
		return map[string]any{
			"protocolVersion": "2024-11-05",
			"serverInfo":      map[string]any{"name": "priorart", "version": "0.1.0"},
			"capabilities":    map[string]any{"tools": map[string]any{}},
		}, nil
	case "tools/list":
		return map[string]any{"tools": s.tools()}, nil
	case "tools/call":
		return s.call(params)
	default:
		return nil, &RPCError{Code: -32601, Message: "method not found: " + method}
	}
}

func (s *Server) call(params map[string]any) (any, *RPCError) {
	name, _ := params["name"].(string)
	args, _ := params["arguments"].(map[string]any)
	ctx := context.Background()
	var payload any
	var err error
	switch name {
	case "prior_art":
		var r candidate.PriorArtResult
		r, err = s.core.PriorArt(ctx, str(args, "query"), str(args, "ecosystem"), intv(args, "limit", 5))
		payload = r
	case "search_own":
		payload, err = s.core.SearchOwn(ctx, str(args, "query"), intv(args, "limit", 10))
	case "search_registry":
		var r candidate.PriorArtResult
		r, err = s.core.PriorArt(ctx, str(args, "query"), str(args, "ecosystem"), intv(args, "limit", 5))
		payload = r
	case "record_decision":
		err = s.core.RecordDecision(candidate.Decision{
			Query: str(args, "query"), Scope: str(args, "ecosystem"),
			Action: str(args, "action"), Choice: str(args, "choice"), Reason: str(args, "reason"),
		})
		payload = map[string]any{"ok": true}
	case "wheels":
		payload = wheels.All()
	default:
		return nil, &RPCError{Code: -32602, Message: "unknown tool: " + name}
	}
	if err != nil {
		return contentResult(fmt.Sprintf("priorart error: %v", err)), nil
	}
	return contentResult(toJSON(payload)), nil
}

func contentResult(text string) map[string]any {
	return map[string]any{"content": []any{map[string]any{"type": "text", "text": text}}}
}

func toJSON(v any) string {
	b, _ := json.Marshal(v)
	return string(b)
}

func str(m map[string]any, k string) string {
	if m == nil {
		return ""
	}
	s, _ := m[k].(string)
	return s
}

func intv(m map[string]any, k string, def int) int {
	if m == nil {
		return def
	}
	if f, ok := m[k].(float64); ok {
		return int(f)
	}
	return def
}

type rpcMessage struct {
	JSONRPC string          `json:"jsonrpc"`
	ID      any             `json:"id,omitempty"`
	Method  string          `json:"method,omitempty"`
	Params  json.RawMessage `json:"params,omitempty"`
	Result  any             `json:"result,omitempty"`
	Error   *RPCError       `json:"error,omitempty"`
}

// Serve reads newline-delimited JSON-RPC messages from r and writes responses to w.
func (s *Server) Serve(r io.Reader, w io.Writer) error {
	sc := bufio.NewScanner(r)
	sc.Buffer(make([]byte, 0, 64*1024), 4*1024*1024)
	for sc.Scan() {
		line := sc.Bytes()
		if len(line) == 0 {
			continue
		}
		var msg rpcMessage
		if err := json.Unmarshal(line, &msg); err != nil {
			continue
		}
		var params map[string]any
		if len(msg.Params) > 0 {
			_ = json.Unmarshal(msg.Params, &params)
		}
		if msg.Method == "notifications/initialized" || msg.Method == "initialized" {
			continue // notification: no response
		}
		res, rpcErr := s.Dispatch(msg.Method, params)
		out := rpcMessage{JSONRPC: "2.0", ID: msg.ID, Result: res, Error: rpcErr}
		b, _ := json.Marshal(out)
		if _, err := fmt.Fprintln(w, string(b)); err != nil {
			return err
		}
	}
	return sc.Err()
}
```
